function renderReports() {
const uData = getUserData(currentUser);
const healthContainer = document.getElementById("healthReportContainer");
healthContainer.innerHTML = "";
sortHistoryByWorkoutDateDesc(uData.health).forEach(h => {
const items = [];
if (h.weight) items.push(`Kilo: ${h.weight}kg`);
if (h.fat) items.push(`Yağ: %${h.fat}`);
if (h.waist) items.push(`Bel: ${h.waist}cm`);
if (h.hip) items.push(`Kalça: ${h.hip}cm`);
if (h.shoulder) items.push(`Omuz: ${h.shoulder}cm`);
if (h.arm) items.push(`Kol: ${h.arm}cm`);
if (h.thigh) items.push(`Bacak: ${h.thigh}cm`);
if (h.neck) items.push(`Boyun: ${h.neck}cm`);
healthContainer.innerHTML += `
<div style="padding:8px 0; border-bottom:1px solid var(--border-color); font-size:13px;">
<strong style="color:var(--primary);">${h.date}:</strong> ${items.join(' | ')}
${h.notes ? `<div style="font-size:11px; color:var(--text-muted); margin-top:2px;">Not: ${h.notes}</div>` : ''}
</div>
`;
});
const exSelect = document.getElementById("reportExerciseSelect");
exSelect.innerHTML = '<option value="">-- Program & Egzersiz Seçin --</option>';
const seenKeys = new Set();
uData.history.forEach(h => {
const progObj = uData.programs.find(p => p.id === h.programId || p.name === h.programName);
const pId = progObj ? progObj.id : (h.programId || "default");
const pName = progObj ? progObj.name : (h.programName || "Genel");
const key = `${pId}|||${h.exerciseName}`;
if (!seenKeys.has(key)) {
seenKeys.add(key);
exSelect.innerHTML += `<option value="${key}">[${pName}] ${h.exerciseName}</option>`;
}
});
}
function renderExerciseReport() {
const selectedVal = document.getElementById("reportExerciseSelect").value;
const container = document.getElementById("exerciseReportContainer");
container.innerHTML = "";
if (!selectedVal) return;
const [targetProgId, targetExName] = selectedVal.split("|||");
const uData = getUserData(currentUser);
const prog = uData.programs.find(p => p.id === targetProgId);
const filtered = sortHistoryByWorkoutDateDesc(
uData.history.filter(h => {
if (h.exerciseName !== targetExName) return false;
if (prog) return isRecordForProgram(h, prog, uData);
return (h.programId || "default") === targetProgId;
})
);
filtered.forEach(h => {
const setStr = h.sets.map((s, i) => `#${i+1}: ${s.weight > 0 ? s.weight+'kg x ' : ''}${s.reps}tekrar`).join('<br>');
container.innerHTML += `
<div style="padding:8px 0; border-bottom:1px solid var(--border-color); font-size:13px;">
<strong style="color:var(--primary);">${h.date}</strong><br>
${setStr}
</div>
`;
});
}
